cmake_minimum_required(VERSION 3.21)
project(BleProbe LANGUAGES CXX)
get_filename_component(SCK_ROOT "../../.." ABSOLUTE)
add_library(SimpleCommKitBle SHARED IMPORTED)
set_target_properties(SimpleCommKitBle PROPERTIES IMPORTED_LOCATION "${BLE_LIB_DIR}/libSimpleCommKitBle.so")
add_library(SimpleCommKitUtil SHARED IMPORTED)
set_target_properties(SimpleCommKitUtil PROPERTIES IMPORTED_LOCATION "${UTIL_LIB_DIR}/libSimpleCommKitUtil.so")
add_library(BleProbe SHARED NativeProbe.cpp)
target_compile_features(BleProbe PRIVATE cxx_std_17)
target_include_directories(BleProbe PRIVATE "${SCK_ROOT}/src/SimpleCommKitBle" "${SCK_ROOT}/src/SimpleCommKitUtil")
target_include_directories(BleProbe PRIVATE "${SCK_ROOT}/examples/SimpleCommKitBle/SimpleCommKitBleCentralExample")
target_link_libraries(BleProbe PRIVATE SimpleCommKitBle SimpleCommKitUtil)
